  Weights: $W(a) = \frac13,$\\
  \phantom{Weights: }$W(\bar a) = \frac23$, \\

\begin{center}
  \begin{tabular}{|l|l|}
  \hline
  	$\bar a \bar b $ & --- \\
  	$\bar a  b $ & $\nicefrac 23$ \\
  	$a \bar b $ & $\nicefrac 13$ \\
  	$a b$ & $\nicefrac 13$\\
  \hline
  \end{tabular}

\text{ModelCount}_W(F) = \frac 43 \pause = \sum_{\vec x\in \set{0,1}^n} \overbrace{F(\vec x)}^{\text{satisfiable 0/1}}  \cdot \overbrace{\prod_{x_i \in \vec x} W(x_i)}^{\text{weight of assignment}}

\end{example}
